//play a sound in microphone

Nsp=44100

tmp=1  //s

defM1,1,(tmp*Nsp)
fillM1,0,tmp
pi=3.1415927
//M0=0.5*sin(2*pi*50*M1)+ 0.3*cos(2*pi*100*M1)
defM0,1,(tmp*Nsp)
recsndM0,Nsp
gfxdim -0.1,(Nsp/6),-20000,20000,5
grid1000
x=1
Trf(x)=M0(1,x)

M1=M0 //to get same mtx size
fillM1,0,tmp // up to 2sec?

f=1/tmp
pi=3.1415927
w=2*pi*f

km=1500  // ------- kmax
defM3,1,km
defM7,1,km
fillM3,0,0
fillM7,0,0
//test fftM:Remove 3 comments
fftM0,3,7
bsr DisplaySpectrum
end

//Below is the FFT using 
//the matrix product
//slower than fftM
k=1
loop:
M3(1,k)= 2/Nsp*M0*trp(cos(k*w*M1))
M7(1,k)= 2/Nsp*M0*trp(sin(k*w*M1))
k=k+1
k<km=>goto loop
bsr DisplaySpectrum
end


DisplaySpectrum:
gfxdim -0.1,km,-3000,3000,1
grid100
x=1
Trf(x)=abs(M3(1,x))
x=1
Trf(x)=abs(M7(1,x))
rts
