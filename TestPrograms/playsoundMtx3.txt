gfxdim -0.5,100,-1.2,1.2,0.05
grid10
Nsp=96000 //samples /s
f=440 //Hz
dt=1/Nsp
duration=2 // seconds
le=duration/dt
pi=3.1415927

defM0,1,le
defM1,1,le

w=2*pi*f
fillM0,0,duration //0 to 2s
M1=0.5*cos(w*M0)+0.5*sin(4*w*M0)
t=10
x=1
loop:
z=x*t
a=M1(1,z)
y=x
x=x+1
z=x*t
b=M1(1,z)
line y,a,x,b,3
x<100 =>goto loop
playsndM1,Nsp,8

loop2:
key(0)==9 =>end
key(0)==1 =>f=f+1
key(0)==2 =>f=f+10
key(0)==3 =>f=f+100
key(0)==4 =>f=f+1000
key(0)==5 =>f=f+10000
goto loop




end
