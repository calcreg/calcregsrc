//Test FFT
Ntot=50000
tmp=1 //s
kmax=100
Nsp=Ntot/tmp
Nyquist=Nsp/2
defM0,1,Ntot
fillM0,0,tmp
pi=3.1415927
f0=25 //Hz //f0<Nyquist!
w=2*pi*f0
M1=cos(w*M0)+0.5*cos(2*w*M0)
//plot
gfxdim 0,Ntot,-5,2,1
grid 10
x=1
Trf(x)=M1(1,x)

//FFT
defM2,1,kmax
k=1
loopfft:
M2(1,k)=M1*trp(cos(2*pi*k*M0))
k=k+1
k<kmax=>goto loopfft

gfxdim 0,kmax,0,Ntot,1
x=1
Trf(x)=abs(M2(1,x))
