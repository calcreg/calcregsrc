//butterworth filters
//You must allow complexe calculation




wmax=0.3E+3
dw=1E+3
R0=1    //ohm start resistance
dR=20  //increment R=R+dR
Rmax=50

C1=3.3E-6 //F
C3=3.3E-6
L2=680E-6//H
L4=560E-6

//C1,L2,C3,L4

gfxdim 0,(wmax-1),-100,100,1
grid100
defM0,1,wmax

R=R0
Start:
bsr Butterworth
R=R+dR
R<Rmax=>goto Start
end


Butterworth:
w=1
loop:
bsr CALCIMPEDANCE
bsr ALPHA
bsr BETA
bsr TRANSFER
M0(1,w)=mod(transf)
w=w+1
w<wmax=>goto loop

//Display in dB Scale
x=1
Trf(x)=200*(M0(1,x))/ln10
rts



//transfer function
CALCIMPEDANCE:
z1=1/(i*C1*w*dw)
z3=1/(i*C3*w*dw)
z2=i*L2*w*dw
z4=i*L4*w*dw
rts
ALPHA:
alpha=R*z1*z2/(z1*z2+R*z2+R*z1)
rts
BETA:
beta=1/(1+(z3+z4+R)/(R+z4)*z2/z3)
rts
TRANSFER:
transf=alpha*beta/((1-alpha*beta/z2)*(R+z4))
rts
