//butterworth filters
//You must allow complexe calculation



//R, C1,L2,C3,L4
//-- R-----L2------L4--------
//         |         |             |
//        C1     C3          R
//         |         |             |
//        =        =           =         
R=100
C1=33E-12 //F
C3=15E-12
L2=68E-9//H
L4=68E-9

//wmax= Nbr Pts calculation
//wdisplayed is W=w*dw
wmax=5E+2
dw=10E+6
//Var C
C0=3E-12
dC=10E-12  //increment R=R+dR
Cmax=100E-12


gfxdim 0,(wmax-1),-100,10,1
grid100
defM0,1,wmax

C=C0
Start:
C1=C //Chg here C1 or C3
bsr Butterworth
C=C+dC
C<Cmax=>goto Start
end


Butterworth:
w=1
loop:
bsr CALCIMPEDANCE
bsr ALPHA
bsr BETA
bsr TRANSFER
M0(1,w)=mod(transf)
w=w+1
w<wmax=>goto loop

//Display in dB Scale
x=1
Trf(x)=20*ln(M0(1,x))/ln10
rts



//transfer function
CALCIMPEDANCE:
z1=1/(i*C1*w*dw)
z3=1/(i*C3*w*dw)
z2=i*L2*w*dw
z4=i*L4*w*dw
rts
ALPHA:
alpha=R*z1*z2/(z1*z2+R*z2+R*z1)
rts
BETA:
beta=1/(1+(z3+z4+R)/(R+z4)*z2/z3)
rts
TRANSFER:
transf=alpha*beta/((1-alpha*beta/z2)*(R+z4))
rts
