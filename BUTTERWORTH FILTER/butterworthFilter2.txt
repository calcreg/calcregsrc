//butterworth filters
//You must allow complexe calculation

wmax=10000
dw=1
R=100 //ohm
C1=10E-9 //10nF
C3=10E-9
L2=330E-9//0.33mH
L4=330E-9

//C1,L2,C3,L4



defM0,1,wmax

w=1
loop:
bsr CALCIMPEDANCE
bsr ALPHA
bsr BETA
bsr TRANSFER
M0(1,w)=mod(transf)
//printM0(1,w)
w=w+1
w<wmax=>goto loop

gfxdim 0,wmax,-10,10,1
grid100
x=1
Trf(x)=M0(1,x)
end



//transfer function
CALCIMPEDANCE:
z1=1/(i*C1*w*dw)
z3=1/(i*C3*w*dw)
z2=i*L2*w*dw
z4=i*L4*w*dw
rts
ALPHA:
alpha=R*z1*z2/(z1*z2+R*z2+R*z1)
rts
BETA:
beta=1/(1+(z3+z4+R)/(R+z4)*z2/z3)
rts
TRANSFER:
transf=alpha*beta/((1-alpha*beta/z2)*(R+z4))
rts
