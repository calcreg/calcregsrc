//butterworth filters
//You must allow complexe calculation


//R, C1,L2,C3,L4,C5

//          --C5--
//         |         |
//-- R-----L2------L4--------
//         |         |             |
//        C1     C3          R
//         |         |             |
//        =        =           =         



wmax=1E+3
dw=10E+6
R0=1    //ohm start resistance
dR=50  //increment R=R+dR
Rmax=400

C1=33E-12 //F
C3=15E-12
L2=68E-9//H
L4=68E-9
C5=1E-12
//C1,L2,C3,L4

gfxdim 0,(wmax-1),-100,10,1
grid100
defM0,1,wmax

R=R0
Start:
bsr Butterworth
R=R+dR
R<Rmax=>goto Start
end


Butterworth:
w=1
loop:
bsr CALCIMPEDANCE
bsr ALPHA
bsr BETA
bsr TRANSFER
M0(1,w)=mod(transf)
w=w+1
w<wmax=>goto loop

//Display in dB Scale
x=1
Trf(x)=20*ln(M0(1,x))/ln10
rts



//transfer function
CALCIMPEDANCE:
z1=1/(i*C1*w*dw)
z3=1/(i*C3*w*dw)
z2=i*L2*w*dw/(1-L2*C5*(w*dw)^2)
z4=i*L4*w*dw
rts
ALPHA:
alpha=R*z1*z2/(z1*z2+R*z2+R*z1)
rts
BETA:
beta=1/(1+(z3+z4+R)/(R+z4)*z2/z3)
rts
TRANSFER:
transf=alpha*beta/((1-alpha*beta/z2)*(R+z4))
rts
