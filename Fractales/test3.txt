//#COMPLEXEON
action0
clrlimit=50
c=-0.00101+i*1.025

xmin=-0.1517779
xmax=-0.1517609
ymin=-0.4187762
ymax=-0.4187689
gfxdim xmin,xmax,ymin,ymax,1

x=xmin
dx=(xmax-xmin)/300
y=ymin
dy=(ymax-ymin)/350
print dy,"dy="
loop:
x=x+dx
bsr calculate
bsr display
x<xmax=>goto loop
x=xmin
y=y+dy
y<ymax=>goto loop
end


display:
line x,y,(x+dx),y,(couleur/2)
rts

calculate:
z=x+i*y
couleur=0
lpfr:
z=z*z+c
couleur=couleur+1
couleur>clrlimit=> goto initclr
mod(z)<2=>goto lpfr
rts

initclr:
couleur=0
rts
