//#COMPLEXEON
action0
clrlimit=50
c=-1.001+i*0.005

xmin=-2
xmax=2
ymin=-1.07
ymax=1.08
gfxdim xmin,xmax,ymin,ymax,1

x=xmin
dx=(xmax-xmin)/150
y=ymin
dy=(ymax-ymin)/300
loop:
x=x+dx
bsr calculate
bsr display
x<xmax=>goto loop
x=xmin
y=y+dy
y<ymax=>goto loop
end


display:
line x,y,(x+dx),y,couleur
rts

calculate:
z=x+i*y
couleur=0
lpfr:
z=z*z+c
couleur=couleur+1
couleur>clrlimit=> goto initclr
mod(z)<2=>goto lpfr
rts

initclr:
couleur=0
rts
