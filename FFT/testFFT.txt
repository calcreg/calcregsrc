f=1/tmp
pi=3.1415927
w=2*pi*f

km=800  // ------- kmax
defM3,1,km
defM7,1,km
fillM3,0,0
fillM7,0,0

k=1
loop:
M3(1,k)= M0*trp(cos(k*w*M1))
M7(1,k)= M0*trp(sin(k*w*M1))
k=k+1
k<km=>goto loop
bsr DisplaySpectrum
end
Nsp=8192 //--------- Nsp file2
defM4,1,Nsp

fillM5,1,km
t=1
dt=tmp/Nsp
loop2:
M4(1,t)= M3*trp(cos(t*dt*w*M5))+M7*trp(sin(t*dt*w*M5))
t=t+1
t<Nsp=>goto loop2

end

jouer:

end
