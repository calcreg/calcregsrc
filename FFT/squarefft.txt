Nsp=4096

tmp=1  //s
createbtn1,130,360,80,20,"Spectrum"
createbtn9,130,380,80,20,"Quit"
defM1,1,Nsp
fillM1,0,tmp
pi=3.1415927
//M0=0.5*sin(2*pi*50*M1)+ 0.3*cos(2*pi*100*M1)
//M0=0.5*sign(-0.9+sin(2*pi*50*M1)) //% inegal
M0=0.5*sign(sin(2*pi*50*M1)) //square

gfxdim -0.1,(Nsp/6),-2,2,5
grid1000
x=1
Trf(x)=M0(1,x)

loopClv:
clavier=key(0)
clavier==1=>goto CalcSpectrum
clavier==9=>goto quit
goto loopClv
quit:
end

CalcSpectrum:
M1=M0 //to get same mtx size
fillM1,0,tmp // up to 2sec?

f=1/tmp
pi=3.1415927
w=2*pi*f

km=800  // ------- kmax
defM3,1,km
defM7,1,km
fillM3,0,0
fillM7,0,0
//test fftM:Remove 3 comments
fftM0,3,7
bsr DisplaySpectrum
end

//Below is the FFT using 
//the matrix product
//slower than fftM
k=1
loop:
M3(1,k)= 2/Nsp*M0*trp(cos(k*w*M1))
M7(1,k)= 2/Nsp*M0*trp(sin(k*w*M1))
k=k+1
k<km=>goto loop
bsr DisplaySpectrum
end


DisplaySpectrum:
gfxdim -10,km,-1,1,1
grid100
x=1
Trf(x)=abs(M3(1,x))
colorgfx3
x=1
Trf(x)=abs(M7(1,x))
rts
